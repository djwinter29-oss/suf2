cmake_minimum_required(VERSION 3.13)

# Select the chip before importing the SDK and its toolchain.
set(SUF2_BOARD rp2040 CACHE STRING "SUF2 target chip: rp2040 or rp2350")
set_property(CACHE SUF2_BOARD PROPERTY STRINGS rp2040 rp2350)

if (SUF2_BOARD STREQUAL "rp2040")
    set(_platform rp2040)
    set(_board pico)
    set(SUF2_DEFAULT_FLASH_SIZE 0x200000)
    set(SUF2_APP_VECTOR_OFFSET 0x100)
    set(SUF2_DEFAULT_PRODUCT_NAME SUF2RP2040)
elseif (SUF2_BOARD STREQUAL "rp2350")
    set(_platform rp2350-arm-s)
    set(_board pico2)
    set(SUF2_DEFAULT_FLASH_SIZE 0x400000)
    set(SUF2_APP_VECTOR_OFFSET 0)
    set(SUF2_DEFAULT_PRODUCT_NAME SUF2RP2350)
else()
    message(FATAL_ERROR "SUF2_BOARD must be rp2040 or rp2350")
endif()

if (CMAKE_PROJECT_NAME AND NOT CMAKE_PROJECT_NAME STREQUAL "suf2_${SUF2_BOARD}")
    message(FATAL_ERROR "Use a separate build directory for each SUF2_BOARD")
endif()
set(PICO_PLATFORM ${_platform} CACHE STRING "Pico SDK platform")
set(PICO_BOARD ${_board} CACHE STRING "Pico SDK board header")
if (NOT PICO_PLATFORM STREQUAL _platform)
    message(FATAL_ERROR "SUF2_BOARD=${SUF2_BOARD} requires PICO_PLATFORM=${_platform}")
endif()
if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE MinSizeRel)
endif()

include(${CMAKE_CURRENT_LIST_DIR}/pico_sdk_import.cmake)
project(suf2_${SUF2_BOARD} C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if (NOT PICO_PLATFORM STREQUAL _platform)
    message(FATAL_ERROR "PICO_BOARD=${PICO_BOARD} does not match SUF2_BOARD=${SUF2_BOARD}")
endif()

set(SUF2_BOARD_NAME ${SUF2_BOARD})
include(${CMAKE_CURRENT_LIST_DIR}/suf2_rp2.cmake)
add_subdirectory(suf2-boot)
add_subdirectory(blink)
